🔹 جملة if البسيطة

JavaScript
let age = 20;

if (age >= 18) {
  console.log("أنت بالغ");
}
📤 النتيجة
أنت بالغ

القاعدة:

  • if = "إذا"
  • الشرط بين ()
  • الكود المنفّذ بين {}

🔸 إضافة else

ماذا لو لم يتحقق الشرط؟ نستخدم else:

JavaScript
let age = 15;

if (age >= 18) {
  console.log("أنت بالغ");
} else {
  console.log("أنت قاصر");
}
📤 النتيجة
أنت قاصر

🔷 else if — شروط متعددة

JavaScript
let grade = 85;

if (grade >= 90) {
  console.log("ممتاز");
} else if (grade >= 80) {
  console.log("جيد جداً");
} else if (grade >= 70) {
  console.log("جيد");
} else if (grade >= 60) {
  console.log("مقبول");
} else {
  console.log("راسب");
}
📤 النتيجة
جيد جداً
💡
مهم: ترتيب الشروط يؤثر! ضع الشروط الصارمة أولاً. في المثال فوق، لو وضعنا >= 60 أولاً ستحقق دائماً.

let age = 25;

⚠️
في JavaScript، العوامل هي &&, ||, !. في Python كانت and, or, not. لا تخلط!

🎯 العامل الثلاثي (Ternary)

اختصار أنيق لـ if/else البسيط:

الطريقة التقليدية
let age = 20;
let status;

if (age >= 18) {
  status = "بالغ";
} else {
  status = "قاصر";
}
Ternary
let age = 20;
let status = age >= 18 ? "بالغ" : "قاصر";
console.log(status);
📤 النتيجة
بالغ

الصيغة: الشرط ? قيمة_إذا_صحيح : قيمة_إذا_خطأ

🎛️ جملة switch

بديل أنيق عند مقارنة قيمة واحدة بعدة احتمالات:

JavaScript
let day = 3;
let name;

switch (day) {
  case 1: name = "الأحد"; break;
  case 2: name = "الاثنين"; break;
  case 3: name = "الثلاثاء"; break;
  case 4: name = "الأربعاء"; break;
  case 5: name = "الخميس"; break;
  default: name = "يوم آخر";
}

console.log(name);
📤 النتيجة
الثلاثاء
⚠️
لا تنسَ break! بدونها سيكمل التنفيذ للحالات التالية (fall-through).

✏️ تمارين تطبيقية

تمرين 1سهل

أنشئ let num = 15; واطبع "زوجي" أو "فردي".

💡 تلميح

استخدم num % 2 === 0.

✅ الحل
let num = 15;
if (num % 2 === 0) {
  console.log("زوجي");
} else {
  console.log("فردي");
}
تمرين 2سهل

أنشئ let score = 75; واطبع "ناجح" إذا كانت ≥ 60، وإلا "راسب".

💡 تلميح

استخدم if و else.

✅ الحل
let score = 75;
if (score >= 60) {
  console.log("ناجح");
} else {
  console.log("راسب");
}
تمرين 3متوسط

أنشئ let age = 20; و let hasTicket = true; واطبع "مسموح" فقط إذا كان العمر ≥ 18 ولديه تذكرة.

💡 تلميح

استخدم &&.

✅ الحل
let age = 20;
let hasTicket = true;
if (age >= 18 && hasTicket) {
  console.log("مسموح");
} else {
  console.log("ممنوع");
}
تمرين 4متوسط

استخدم Ternary لطباعة "حار" إذا كانت الحرارة > 30، وإلا "معتدل".

💡 تلميح

الصيغة: condition ? a : b

✅ الحل
let temp = 35;
let feeling = temp > 30 ? "حار" : "معتدل";
console.log(feeling);

📝 اختبار سريع

1. ما العامل المنطقي "و" في JavaScript؟

أ) and
ب) &&
ج) &
د) ++

2. ما الصيغة الصحيحة لجملة if؟

أ) if x > 5 { }
ب) if (x > 5) { }
ج) if x > 5 then { }
د) if [x > 5] { }

3. ما نتيجة 5 > 3 ? "أ" : "ب"؟

أ) "أ"
ب) "ب"
ج) خطأ
د) undefined

4. ماذا يحدث لو نسيت break في switch؟

أ) لا شيء
ب) خطأ
ج) يكمل للحالات التالية
د) يتوقف التطبيق

🎓 خلاصة الدرس

  • ✅ if لاتخاذ قرار
  • ✅ else للبديل
  • ✅ else if لعدة احتمالات
  • ✅ العوامل: &&, ||, !
  • ✅ Ternary: cond ? a : b
  • ✅ switch لعدة حالات — لا تنسَ break
