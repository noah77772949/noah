🤔 لماذا نحتاج الدوال؟

تخيّل أنك تريد حساب مساحة دائرة 10 مرات في برنامجك. هل تكتب نفس الصيغة 10 مرات؟

❌ بدون دوال (ممل)
let r1 = 5;
let area1 = 3.14 * r1 * r1;
console.log(area1);

let r2 = 10;
let area2 = 3.14 * r2 * r2;
console.log(area2);

let r3 = 7;
let area3 = 3.14 * r3 * r3;
console.log(area3);
// ... 7 مرات أخرى!

مع الدوال، اكتب الصيغة مرة واحدة واستخدمها متى شئت:

✅ مع الدوال (أنيق)
function circleArea(r) {
  return 3.14 * r * r;
}

console.log(circleArea(5));
console.log(circleArea(10));
console.log(circleArea(7));
🎓
قاعدة برمجية: لو نسخت كوداً مرتين، اعمل منه دالة. هذا مبدأ DRY: Don't Repeat Yourself.

🔨 إنشاء أول دالة

الصيغة الأساسية:

JavaScript
function functionName() {
  // الكود هنا
}

مثال عملي:

JavaScript
function sayHello() {
  console.log("مرحباً بالعالم!");
}

// استدعاء الدالة
sayHello();
sayHello();
sayHello();
📤 النتيجة
مرحباً بالعالم!
مرحباً بالعالم!
مرحباً بالعالم!
⚠️
لا تنسَ الأقواس! كتابة sayHello بدون () تعني "أشر إلى الدالة" وليس "نفّذها". الفرق ضخم.

📥 الوسائط (Parameters)

يمكنك تمرير قيم للدالة لتستخدمها:

JavaScript
function greet(name) {
  console.log("مرحباً " + name);
}

greet("أحمد");
greet("سارة");
greet("محمد");
📤 النتيجة
مرحباً أحمد
مرحباً سارة
مرحباً محمد

عدة وسائط

JavaScript
function introduce(name, age, city) {
  console.log(`أنا ${name} من ${city} وعمري ${age}`);
}

introduce("أحمد", 25, "دمشق");
introduce("سارة", 30, "بيروت");
📤 النتيجة
أنا أحمد من دمشق وعمري 25
أنا سارة من بيروت وعمري 30

قيم افتراضية

يمكنك إعطاء قيمة افتراضية للوسيط:

JavaScript
function greet(name = "زائر") {
  console.log("مرحباً " + name);
}

greet("أحمد");
greet();  // يستخدم القيمة الافتراضية
📤 النتيجة
مرحباً أحمد
مرحباً زائر

📤 إرجاع القيم (return)

حتى الآن دوالنا تطبع فقط. لكن الدوال الأقوى هي التي تُرجع قيمة.

JavaScript
function add(a, b) {
  return a + b;
}

let result = add(5, 3);
console.log(result);
console.log(add(10, 20));
console.log(add(100, 200));
📤 النتيجة
8
30
300

الفرق بين console.log و return

console.logreturn
الوظيفةيُظهر على الشاشةيُعيد قيمة للكود
يمكن استخدامها في حساب؟لانعم
تُوقف الدالة؟لانعم
JavaScript
// خطأ شائع: يطبع لكن لا يُرجع
function wrongAdd(a, b) {
  console.log(a + b);
}
let x = wrongAdd(5, 3);
console.log("x =", x);  // x = undefined!

// صحيح
function rightAdd(a, b) {
  return a + b;
}
let y = rightAdd(5, 3);
console.log("y =", y);  // y = 8
💎
return تُوقف تنفيذ الدالة فوراً. أي كود بعدها لا يُنفَّذ.

🎯 Arrow Functions

JavaScript الحديثة لديها طريقة أقصر لكتابة الدوال:

الطريقة التقليدية
function add(a, b) {
  return a + b;
}
Arrow Function
const add = (a, b) => {
  return a + b;
};
أقصر — بسطر واحد
const add = (a, b) => a + b;

القاعدة: إذا كانت الدالة تُرجع قيمة واحدة فقط، يمكن الاستغناء عن {} و return.

أمثلة

JavaScript
// تقليدي
function square(x) { return x * x; }

// Arrow — نفس الشيء
const square2 = x => x * x;

// بدون وسائط
const sayHi = () => console.log("مرحباً!");

// عدة وسائط
const multiply = (a, b) => a * b;

console.log(square2(5));
sayHi();
console.log(multiply(3, 4));
📤 النتيجة
25
مرحباً!
12
🎓
Arrow Functions اختراع حديث (ES6, 2015). معظم كود JavaScript الحديث يستخدمها. تعلّمها مبكراً يجعل كودك أنيقاً.

🐛 أخطاء شائعة

خطأ 1: نسيان ()

❌ خطأ
function sayHi() {
  console.log("مرحبا");
}
sayHi;  // لا يفعل شيء!

الحل: sayHi();

خطأ 2: عدم إرجاع القيمة

❌ خطأ
function double(x) {
  x * 2;  // نسيان return
}
console.log(double(5));  // undefined

الحل: أضف return.

خطأ 3: كود بعد return

❌ خطأ
function test() {
  return 5;
  console.log("لن يُنفَّذ");
}

السبب: return تُوقف التنفيذ فوراً.

✏️ تمارين تطبيقية

تمرين 1 سهل

اكتب دالة welcome() تطبع "أهلاً وسهلاً". ثم استدعها مرتين.

💡 تلميح

استخدم function ثم console.log.

✅ الحل 
function welcome() {
  console.log("أهلاً وسهلاً");
}
welcome();
welcome();
تمرين 2 سهل

اكتب دالة greet(name) تطبع "مرحباً [الاسم]". جرّبها مع 3 أسماء.

💡 تلميح

استخدم template literal: `` `مرحباً ${name}` ``.

✅ الحل 
function greet(name) {
  console.log(`مرحباً ${name}`);
}
greet("أحمد");
greet("سارة");
greet("محمد");
تمرين 3 متوسط

اكتب دالة square(x) تُرجع مربع العدد. استخدمها لطباعة مربعات 3، 5، 7.

💡 تلميح

استخدم return x * x;.

✅ الحل 
function square(x) {
  return x * x;
}
console.log(square(3));
console.log(square(5));
console.log(square(7));
تمرين 4 متوسط

اكتب دالة max(a, b) تُرجع الأكبر من العددين.

💡 تلميح

استخدم if أو العامل الثلاثي a > b ? a : b.

✅ الحل 
function max(a, b) {
  if (a > b) return a;
  return b;
}
console.log(max(10, 20));
console.log(max(100, 50));
تمرين 5 متقدم

أعد كتابة دالة square كـ Arrow Function في سطر واحد.

💡 تلميح

الصيغة: const name = (params) => expr;

✅ الحل 
const square = x => x * x;
console.log(square(6));
console.log(square(10));

📝 اختبار سريع

1. كيف تُعرِّف دالة بسيطة؟

أ) def myFunc()
ب) function myFunc()
ج) func myFunc()
د) define myFunc()

2. ما الفرق بين console.log و return؟

أ) لا فرق
ب) console.log يعرض، return يُرجع قيمة
ج) return أسرع
د) console.log للطباعة فقط

3. ما نتيجة function f(x) { x + 1; } f(5)؟

أ) 6
ب) 5
ج) undefined
د) خطأ

4. ما الصيغة الصحيحة للـ Arrow Function؟

أ) function(x) => x * 2
ب) x => x * 2
ج) (x) -> x * 2
د) lambda x: x * 2

5. ماذا يحدث بعد تنفيذ return؟

أ) يكمل الكود
ب) يُعيد الطباعة
ج) يتوقف تنفيذ الدالة فوراً
د) يُعيد تشغيل الدالة

🎓 خلاصة الدرس

  • ✅ الدوال تمنع تكرار الكود
  • ✅ الصيغة: function name(params) { ... }
  • ✅ الوسائط تُمرَّر بين الأقواس
  • ✅ return تُرجع قيمة وتوقف التنفيذ
  • ✅ Arrow Functions: const f = x => x * 2;
  • ✅ القيم الافتراضية: function f(x = 5)
